\begin{frame}{El problema}
\framesubtitle{Desperdicio digital informativo}

\begin{columns}[T]
\column{0.5\textwidth}
Para informarse en Bolivia hoy:
\begin{itemize}
    \item Abrir múltiples sitios de noticias
    \item Comparar titulares repetidos
    \item Hacer \textit{scroll} en redes sociales
    \item Consumir páginas enteras con publicidad
\end{itemize}

\column{0.5\textwidth}
\centering
\fcolorbox{verde-ecobrief}{verde-ecobrief!5!white}{\parbox{0.85\textwidth}{\centering\bfseries\large\color{azul-profundo} \alert{Desperdicio digital}\\[4pt] tiempo perdido\\ datos móviles innecesarios\\ sin fuente trazable}}
\end{columns}
\end{frame}

\vspace{0.4cm}
\begin{columns}[T]
\column{0.5\textwidth}
\begin{enumerate}
    \item Iniciar corrida (manual o \textit{scheduler})
    \item Verificar caché de resúmenes
    \item Recolectar noticias nuevas
    \item Filtrar artículos sin contenido
\end{enumerate}
\column{0.5\textwidth}
\begin{enumerate}\setcounter{enumi}{4}
    \item Deduplicar (URL + título + huella)
    \item Clasificar por categoría
    \item Rankear por relevancia
    \item Generar briefs con IA
\end{enumerate}
\end{columns}
\end{frame}

\begin{frame}{Eficiencia en el uso de IA}

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{Antes de la IA}
\begin{enumerate}
    \item \alert{Filtrar} --- descartar sin contenido
    \item \alert{Deduplicar} --- evitar repetidos
    \item \alert{Rankear} --- solo lo relevante
    \item \alert{Limitar} --- máximo por categoría
\end{enumerate}
\end{block}

\column{0.5\textwidth}
\begin{block}{Durante la IA}
\begin{enumerate}\setcounter{enumi}{4}
    \item \alert{Cachear} --- no reprocesar
    \item \alert{Modelos pequeños} --- tareas simples
    \item \alert{Control cruzado} --- sin duplicados
\end{enumerate}
\end{block}
\end{columns}

\begin{itemize}
    \item \alert{Problema real:} el exceso de información redundante es desperdicio digital

    \item \alert{Resultados:} 174 noticias $\to$ 39 briefs (77.6\% de reducción)
    \item \alert{IA responsable:} solo después de filtrar, deduplicar y priorizar
    \item \alert{Bajo costo:} menos de USD 110/año de infraestructura
    \item \alert{Green Tech:} menos páginas, menos datos, menos ruido
